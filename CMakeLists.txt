cmake_minimum_required(VERSION 3.24)
project(pikapikaengine VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(pikapika_core core/src/core.cpp core/src/portable.cpp core/src/dynamics.cpp core/src/hydrostatics.cpp core/src/continuous.cpp core/src/interval.cpp core/src/accelerated.cpp core/src/optics.cpp core/src/continuous_world.cpp core/src/optical_geometry.cpp core/src/polarization.cpp core/src/dispersion.cpp core/src/optical_bounds.cpp core/src/support.cpp core/src/optical_route.cpp core/src/optical_selection.cpp core/src/optical_spot.cpp core/src/constrained.cpp core/src/trajectory_guard.cpp core/src/supported_world.cpp core/src/simulation_api.cpp)
target_include_directories(pikapika_core PUBLIC core/include)
if(MSVC)
  target_compile_options(pikapika_core PRIVATE /W4 /permissive- /fp:precise)
else()
  target_compile_options(pikapika_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
# The enclosure kernel depends on binary64 operations and gradual underflow;
# keep its build contract explicit even if other simulation sources change flags.
if(MSVC)
  set_source_files_properties(core/src/interval.cpp core/src/accelerated.cpp core/src/optics.cpp core/src/continuous_world.cpp core/src/optical_geometry.cpp core/src/polarization.cpp core/src/dispersion.cpp core/src/optical_bounds.cpp core/src/support.cpp core/src/optical_route.cpp core/src/optical_selection.cpp core/src/optical_spot.cpp core/src/constrained.cpp core/src/trajectory_guard.cpp core/src/supported_world.cpp PROPERTIES COMPILE_OPTIONS "/fp:strict")
else()
  set_source_files_properties(core/src/interval.cpp core/src/accelerated.cpp core/src/optics.cpp core/src/continuous_world.cpp core/src/optical_geometry.cpp core/src/polarization.cpp core/src/dispersion.cpp core/src/optical_bounds.cpp core/src/support.cpp core/src/optical_route.cpp core/src/optical_selection.cpp core/src/optical_spot.cpp core/src/constrained.cpp core/src/trajectory_guard.cpp core/src/supported_world.cpp PROPERTIES COMPILE_OPTIONS "-fno-fast-math;-ffp-contract=off;-frounding-math")
endif()
add_executable(pikapika tools/pikapika.cpp)
target_link_libraries(pikapika PRIVATE pikapika_core)
add_executable(pikapika-dynamics tools/dynamics_bench.cpp)
target_link_libraries(pikapika-dynamics PRIVATE pikapika_core)
add_executable(pikapika-buoyancy tools/buoyancy_bench.cpp)
target_link_libraries(pikapika-buoyancy PRIVATE pikapika_core)
add_executable(pikapika-simulation-probe tools/simulation_probe.cpp)
target_link_libraries(pikapika-simulation-probe PRIVATE pikapika_core)
add_executable(pikapika-supported-probe tools/supported_probe.cpp)
target_link_libraries(pikapika-supported-probe PRIVATE pikapika_core)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  OUTPUT_VARIABLE PP_DYN_REVISION_BASE OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT PP_DYN_REVISION_BASE)
  set(PP_DYN_REVISION_BASE "unavailable")
endif()
execute_process(COMMAND git status --porcelain --untracked-files=normal WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  OUTPUT_VARIABLE PP_DYN_WORKTREE_STATUS ERROR_QUIET)
if(PP_DYN_WORKTREE_STATUS)
  set(PP_DYN_DIRTY 1)
else()
  set(PP_DYN_DIRTY 0)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  core/src/dynamics.cpp core/include/pikapika/dynamics.hpp tools/dynamics_bench.cpp core/src/core.cpp core/include/pikapika/core.hpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/src/dynamics.cpp" PP_DYN_SOURCE_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/include/pikapika/dynamics.hpp" PP_DYN_HEADER_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/tools/dynamics_bench.cpp" PP_DYN_BENCH_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/src/core.cpp" PP_DYN_CORE_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/include/pikapika/core.hpp" PP_DYN_CORE_HEADER_HASH)
target_compile_definitions(pikapika-dynamics PRIVATE
  PP_DYN_REVISION_BASE="${PP_DYN_REVISION_BASE}" PP_DYN_DIRTY=${PP_DYN_DIRTY}
  PP_DYN_SOURCE_HASH="${PP_DYN_SOURCE_HASH}" PP_DYN_HEADER_HASH="${PP_DYN_HEADER_HASH}"
  PP_DYN_BENCH_HASH="${PP_DYN_BENCH_HASH}" PP_DYN_CORE_HASH="${PP_DYN_CORE_HASH}"
  PP_DYN_CORE_HEADER_HASH="${PP_DYN_CORE_HEADER_HASH}"
  PP_DYN_COMPILER_ID="${CMAKE_CXX_COMPILER_ID}" PP_DYN_COMPILER_VERSION="${CMAKE_CXX_COMPILER_VERSION}"
  PP_DYN_BUILD_CONFIG="$<CONFIG>")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  core/src/hydrostatics.cpp core/include/pikapika/hydrostatics.hpp tools/buoyancy_bench.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/src/hydrostatics.cpp" PP_BUOY_SOURCE)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/include/pikapika/hydrostatics.hpp" PP_BUOY_HEADER)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/tools/buoyancy_bench.cpp" PP_BUOY_BENCH)
target_compile_definitions(pikapika-buoyancy PRIVATE
  PP_BUOY_REVISION="${PP_DYN_REVISION_BASE}" PP_BUOY_DIRTY=${PP_DYN_DIRTY}
  PP_BUOY_SOURCE="${PP_BUOY_SOURCE}" PP_BUOY_HEADER="${PP_BUOY_HEADER}" PP_BUOY_BENCH="${PP_BUOY_BENCH}"
  PP_BUOY_DYNAMICS="${PP_DYN_SOURCE_HASH}"
  PP_BUOY_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}" PP_BUOY_CONFIG="$<CONFIG>")
if(WIN32)
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE PIKAPIKA_REVISION_BASE OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(NOT PIKAPIKA_REVISION_BASE)
    set(PIKAPIKA_REVISION_BASE "unavailable")
  endif()
  execute_process(COMMAND git status --porcelain --untracked-files=no WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    OUTPUT_VARIABLE PIKAPIKA_WORKTREE_STATUS ERROR_QUIET)
  if(PIKAPIKA_WORKTREE_STATUS)
    set(PIKAPIKA_WORKTREE_DIRTY 1)
  else()
    set(PIKAPIKA_WORKTREE_DIRTY 0)
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    core/src/core.cpp platform/windows/viewer.cpp)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/src/core.cpp" PIKAPIKA_CORE_SOURCE_SHA256)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/platform/windows/viewer.cpp" PIKAPIKA_VIEWER_SOURCE_SHA256)
  add_executable(pikapika-viewer WIN32 platform/windows/viewer.cpp)
  target_link_libraries(pikapika-viewer PRIVATE pikapika_core d3d12 dxgi d3dcompiler user32 shell32)
  target_compile_definitions(pikapika-viewer PRIVATE UNICODE _UNICODE NOMINMAX
    PIKAPIKA_CODE_REVISION_BASE="${PIKAPIKA_REVISION_BASE}"
    PIKAPIKA_WORKTREE_DIRTY=${PIKAPIKA_WORKTREE_DIRTY}
    PIKAPIKA_CORE_SOURCE_SHA256="${PIKAPIKA_CORE_SOURCE_SHA256}"
    PIKAPIKA_VIEWER_SOURCE_SHA256="${PIKAPIKA_VIEWER_SOURCE_SHA256}")
endif()
