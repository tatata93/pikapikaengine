OUT=ROOT/'reports/world-static-mesh-flight-controlled-tick-copy-preparation/run-001'
PROBE=OUT/'probe.cpp'
LIB=ROOT/'build/wfsm1/Release/pikapika_core.lib'
def need(ok,text):
    if not ok:raise ValueError(text)
def stamp(path):
    raw=path.read_bytes();return {'path':path.relative_to(ROOT).as_posix(),'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
def write(path,value):
    with path.open('x',encoding='utf-8',newline='\n') as f:json.dump(value,f,indent=2,allow_nan=False);f.write('\n')
helper=runpy.run_path(str(ROOT/'reports/world-static-mesh-flight-advance-recipes/check-configured-prototype.py'),run_name='readonly_compiler_facts')

cmake='\n'.join(['cmake_minimum_required(VERSION 3.24)','project(WorldCopyPreparation LANGUAGES CXX)',
    'set(CMAKE_CXX_STANDARD 20)','set(CMAKE_CXX_STANDARD_REQUIRED ON)','set(CMAKE_CXX_EXTENSIONS OFF)',
    'set(CMAKE_INTERPROCEDURAL_OPTIMIZATION FALSE)','set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)',helper['FACTS'],
    'add_library(current_core STATIC IMPORTED)','set_target_properties(current_core PROPERTIES IMPORTED_LOCATION '+quote(LIB)+')',
    'add_executable(copy_probe '+quote(PROBE)+')','target_include_directories(copy_probe PRIVATE '+quote(ROOT/'core/include')+')',
    'target_link_libraries(copy_probe PRIVATE current_core)','target_compile_options(copy_probe PRIVATE /W4 /WX /permissive- /fp:strict /GL-)',
    'target_link_options(copy_probe PRIVATE /LTCG:OFF)','set_property(TARGET copy_probe PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)'])+'\n'
(project/'CMakeLists.txt').write_text(cmake,encoding='utf-8',newline='\n')
(OUT/'CMakeLists.txt.txt').write_bytes((project/'CMakeLists.txt').read_bytes())

for name in ['native','wfa2','wfat1','wfa3','wfs1','wfs2','wfsf1','wfsb1','wfsb2','wfsm1']:

    if p.is_relative_to(ROOT/'core'):
        need(before[p.relative_to(ROOT).as_posix()]==baseline[p.relative_to(ROOT).as_posix()],'Current core source differs from imported-library build inputs')

commands=[]
def run(name,args,timeout=240):
    a=identities();start=time.monotonic();code=None;stdout=b'';stderr=b''
    try:
        p=subprocess.run(list(map(str,args)),cwd=ROOT,capture_output=True,timeout=timeout)
        code,stdout,stderr=p.returncode,p.stdout,p.stderr
    except subprocess.TimeoutExpired as e:stdout=e.stdout or b'';stderr=(e.stderr or b'')+str(e).encode()
    except BaseException as e:stderr=str(e).encode()
    (OUT/(name+'.stdout.txt')).write_bytes(stdout);(OUT/(name+'.stderr.txt')).write_bytes(stderr)
    b=identities();row={'name':name,'argv':list(map(str,args)),'cwd':str(ROOT),'exitCode':code,'seconds':time.monotonic()-start,
        'sourceStable':a==b,'sourceStateSha256Before':hashlib.sha256(json.dumps(a,sort_keys=True).encode()).hexdigest(),
        'sourceStateSha256After':hashlib.sha256(json.dumps(b,sort_keys=True).encode()).hexdigest(),
        'stdout':stamp(OUT/(name+'.stdout.txt')),'stderr':stamp(OUT/(name+'.stderr.txt'))}
    commands.append(row);write(OUT/(name+'.argv.json'),row);print(json.dumps(row),flush=True)
    if code!=0:print(stdout.decode(errors='replace'));print(stderr.decode(errors='replace'),file=sys.stderr)
    need(code==0 and a==b,'Actual command failed or protected source changed: '+name)
    return stdout

        need(re.search(r'(?:^|\s)'+re.escape(token)+r'(?=\s|$)',text,re.I),'Actual strict compile missing '+token)
