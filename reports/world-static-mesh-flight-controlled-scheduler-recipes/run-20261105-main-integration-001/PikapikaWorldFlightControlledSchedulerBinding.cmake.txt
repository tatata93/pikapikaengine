# Independent per-tick packet scheduler configuration. This helper does not alter
# PP_WF_* or the seven-source core identity. Artifact/instruction acceptance is
# separate; a declared profile is not a self-measured executable certificate.
include_guard(GLOBAL)

function(_pikapika_controlled_scheduler_json_quote output value)
  string(REPLACE "\\" "\\\\" escaped "${value}")
  string(REPLACE "\"" "\\\"" escaped "${escaped}")
  string(REPLACE "\n" "\\n" escaped "${escaped}")
  string(REPLACE "\r" "\\r" escaped "${escaped}")
  string(REPLACE "\t" "\\t" escaped "${escaped}")
  set(${output} "\"${escaped}\"" PARENT_SCOPE)
endfunction()


function(_pikapika_packet_config_id output expression configuration)
  set(prefix "$<$<CONFIG:${configuration}>:")
  string(FIND "${expression}" "${prefix}" at)
  if(at LESS 0)
    message(FATAL_ERROR "Missing upstream configuration: ${configuration}")
  endif()
  string(LENGTH "${prefix}" length)
  math(EXPR start "${at}+${length}")
  string(SUBSTRING "${expression}" ${start} 64 identity)
  math(EXPR stop "${start}+64")
  string(SUBSTRING "${expression}" ${stop} 1 close)
  if(NOT identity MATCHES "^[0-9a-f]+$" OR NOT close STREQUAL ">")
    message(FATAL_ERROR "Malformed upstream configuration identity")
  endif()
  set(${output} "${identity}" PARENT_SCOPE)
endfunction()

function(_pikapika_packet_upstream output profile_output target macro configurations)
  if(NOT TARGET "${target}")
    message(FATAL_ERROR "Missing upstream target")
  endif()
  get_target_property(definitions "${target}" COMPILE_DEFINITIONS)
  set(count 0)
  set(profile_count 0)
  set(profile 0)
  foreach(definition IN LISTS definitions)
    set(prefix "${macro}_BUILD_ID=\"")
    string(FIND "${definition}" "${prefix}" at)
    if(at EQUAL 0)
      math(EXPR count "${count}+1")
      string(LENGTH "${prefix}" length)
      string(LENGTH "${definition}" total)
      math(EXPR body_length "${total}-${length}-1")
      string(SUBSTRING "${definition}" ${length} ${body_length} expression)
      string(SUBSTRING "${definition}" -1 1 quote)
      if(NOT quote STREQUAL "\"")
        message(FATAL_ERROR "Upstream identity closing quote missing")
      endif()
    endif()
    foreach(candidate IN ITEMS 0 1)
      if(definition STREQUAL "${macro}_PROFILE=$<IF:$<CONFIG:Release>,${candidate},0>")
        math(EXPR profile_count "${profile_count}+1")
        set(profile "${candidate}")
      endif()
    endforeach()
  endforeach()
  if(NOT count EQUAL 1 OR NOT profile_count EQUAL 1)
    message(FATAL_ERROR "Unknown or duplicate upstream binding definition")
  endif()
  set(rebuilt "")
  foreach(configuration IN LISTS configurations)
    _pikapika_packet_config_id(identity "${expression}" "${configuration}")
    string(APPEND rebuilt "$<$<CONFIG:${configuration}>:${identity}>")
  endforeach()
  if(NOT rebuilt STREQUAL expression)
    message(FATAL_ERROR "Upstream identity expression has unexpected material")
  endif()
  set(${output} "${expression}" PARENT_SCOPE)
  set(${profile_output} "${profile}" PARENT_SCOPE)
endfunction()

# The caller supplies existing core, controlled scheduler and clock targets. Upstream identity comes only from
# existing PP_WF_BUILD_ID/PP_WF_PROFILE configuration, never a runtime unlock.
# Missing upstream configuration builds the genuine default-refusal variant.
function(pikapika_add_world_flight_controlled_scheduler target core_target controlled_target clock_target)
  cmake_parse_arguments(PP_PACKET "" "PROFILE_STEM" "" ${ARGN})
  if(PP_PACKET_UNPARSED_ARGUMENTS OR PP_PACKET_KEYWORDS_MISSING_VALUES)
    message(FATAL_ERROR "Unknown/incomplete controlled_scheduler helper arguments")
  endif()
  if(TARGET "${target}" OR NOT TARGET "${core_target}")
    message(FATAL_ERROR "Controlled scheduler needs a new target and an existing core target")
  endif()
  if(NOT PP_PACKET_PROFILE_STEM)
    set(PP_PACKET_PROFILE_STEM "world-flight-controlled-scheduler-binding")
  endif()
  if(NOT PP_PACKET_PROFILE_STEM MATCHES "^[A-Za-z0-9][A-Za-z0-9_.-]*$")
    message(FATAL_ERROR "Controlled scheduler profile stem must be one safe filename component")
  endif()
  get_filename_component(root "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/.." ABSOLUTE)
  set(header "core/include/pikapika/world_static_mesh_flight_controlled_scheduler.hpp")
  set(source "core/src/world_static_mesh_flight_controlled_scheduler.cpp")
  file(SHA256 "${root}/${header}" header_hash)
  file(SHA256 "${root}/${source}" source_hash)
  file(SHA256 "${CMAKE_CURRENT_FUNCTION_LIST_FILE}" helper_hash)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${root}/${header}" "${root}/${source}" "${CMAKE_CURRENT_FUNCTION_LIST_FILE}")
  set(upstream_id "")
  set(upstream_profile 0)
  if(DEFINED PP_WF_BUILD_ID)
    string(LENGTH "${PP_WF_BUILD_ID}" upstream_length)
    if(NOT upstream_length EQUAL 64 OR NOT PP_WF_BUILD_ID MATCHES "^[0-9a-f]+$")
      message(FATAL_ERROR "Malformed upstream seven-source identity")
    endif()
    set(upstream_id "${PP_WF_BUILD_ID}")
  endif()
  if(DEFINED PP_WF_PROFILE)
    if(NOT "${PP_WF_PROFILE}" MATCHES "^[01]$")
      message(FATAL_ERROR "Unknown upstream seven-source profile")
    endif()
    set(upstream_profile "${PP_WF_PROFILE}")
  endif()
  set(known_target 0)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$" AND NOT CMAKE_CROSSCOMPILING)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" AND MSVC_VERSION EQUAL 1951 AND
       NOT CMAKE_VS_PLATFORM_NAME MATCHES "[Aa][Rr][Mm]" AND NOT CMAKE_GENERATOR_PLATFORM MATCHES "[Aa][Rr][Mm]")
      set(known_target 1)
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 13 AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 14)
      set(known_target 1)
    endif()
  endif()
  set(eligible 0)
  if(known_target AND upstream_profile EQUAL 1 AND NOT upstream_id STREQUAL "")
    set(eligible 1)
  endif()
  if(MSVC)
    set(options /fp:strict /GL- /W4 /WX /permissive-)
    set(link_options /LTCG:OFF)
  else()
    set(options -fno-fast-math -ffp-contract=off -frounding-math -fno-lto -Wall -Wextra -Wpedantic -Werror)
    if(known_target)
      list(APPEND options -fexcess-precision=standard -mfpmath=sse -msse2)
    endif()
    set(link_options -fno-lto)
  endif()
  add_library("${target}" STATIC "${root}/${header}" "${root}/${source}")
  target_include_directories("${target}" PUBLIC "${root}/core/include")
  target_link_libraries("${target}" PUBLIC "${controlled_target}" "${clock_target}" "${core_target}")
  target_compile_options("${target}" PRIVATE ${options})
  target_link_options("${target}" INTERFACE ${link_options})
  set_property(TARGET "${target}" PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
  set_property(TARGET "${target}" PROPERTY PIKAPIKA_CONTROLLED_SCHEDULER_SOURCE_SET "world-static-mesh-flight-controlled-scheduler-v1")
  set_property(TARGET "${target}" PROPERTY PIKAPIKA_CONTROLLED_SCHEDULER_PROFILE_STEM "${PP_PACKET_PROFILE_STEM}")

  set(configurations "${CMAKE_CONFIGURATION_TYPES}")
  if(NOT configurations)
    if(NOT CMAKE_BUILD_TYPE)
      message(FATAL_ERROR "Controlled scheduler requires an explicit build configuration")
    endif()
    set(configurations "${CMAKE_BUILD_TYPE}")
  endif()
  # Consume exactly the existing helper definitions; never mutate their files or
  # targets to export identities. Unknown definition shapes fail closed.
  get_target_property(controlled_set "${controlled_target}" PIKAPIKA_CONTROLLED_TICK_SOURCE_SET)
  get_target_property(clock_set "${clock_target}" PIKAPIKA_SCHEDULER_SOURCE_SET)
  if(NOT controlled_set STREQUAL "world-static-mesh-flight-controlled-tick-v1" OR
     NOT clock_set STREQUAL "world-static-mesh-flight-scheduler-v1")
    message(FATAL_ERROR "Controlled scheduler requires the genuine existing helper targets")
  endif()
  _pikapika_packet_upstream(own_controlled_expression own_controlled_profile "${controlled_target}" PIKAPIKA_CONTROLLED_TICK "${configurations}")
  _pikapika_packet_upstream(own_clock_expression own_clock_profile "${clock_target}" PIKAPIKA_FLIGHT_SCHEDULER "${configurations}")
  if(NOT own_controlled_profile OR NOT own_clock_profile)
    set(eligible 0)
  endif()
  set(flags_json "[")
  set(separator "")
  foreach(option IN LISTS options)
    _pikapika_controlled_scheduler_json_quote(encoded "${option}")
    string(APPEND flags_json "${separator}${encoded}")
    set(separator ",")
  endforeach()
  string(APPEND flags_json "]")
  set(identity_expression "")
  foreach(configuration IN LISTS configurations)
    _pikapika_packet_config_id(controlled_id "${own_controlled_expression}" "${configuration}")
    _pikapika_packet_config_id(clock_id "${own_clock_expression}" "${configuration}")
    string(TOUPPER "${configuration}" upper_config)
    set(profile 0)
    if(configuration STREQUAL "Release" AND eligible)
      set(profile 1)
    endif()
    # Explicitly versioned JSON material; final library hash is deliberately not
    # used here (that would be cyclic). It belongs in post-build provenance.
    set(material "{\"identitySchema\":1,\"sourceSet\":\"world-static-mesh-flight-controlled-scheduler-v1\"")
    foreach(pair IN ITEMS
        "configuration|${configuration}" "compilerId|${CMAKE_CXX_COMPILER_ID}" "compilerVersion|${CMAKE_CXX_COMPILER_VERSION}"
        "compilerPath|${CMAKE_CXX_COMPILER}" "system|${CMAKE_SYSTEM_NAME}" "architecture|${CMAKE_SYSTEM_PROCESSOR}"
        "generator|${CMAKE_GENERATOR}" "generatorPlatform|${CMAKE_GENERATOR_PLATFORM}" "pointerBytes|${CMAKE_SIZEOF_VOID_P}"
        "globalFlags|${CMAKE_CXX_FLAGS}" "configurationFlags|${CMAKE_CXX_FLAGS_${upper_config}}"
        "runtimeLibrary|${CMAKE_MSVC_RUNTIME_LIBRARY}" "upstreamAdvanceBuildId|${upstream_id}" "helperSha256|${helper_hash}"
        "controlledTickBuildId|${controlled_id}" "clockBuildId|${clock_id}")
      string(FIND "${pair}" "|" at)
      string(SUBSTRING "${pair}" 0 ${at} key)
      math(EXPR start "${at}+1")
      string(SUBSTRING "${pair}" ${start} -1 value)
      _pikapika_controlled_scheduler_json_quote(encoded "${value}")
      string(APPEND material ",\"${key}\":${encoded}")
    endforeach()
    string(APPEND material ",\"profile\":${profile},\"upstreamProfile\":${upstream_profile},\"compileOptions\":${flags_json},\"IPO\":false,\"sources\":[{\"path\":\"${header}\",\"sha256\":\"${header_hash}\",\"flags\":[]},{\"path\":\"${source}\",\"sha256\":\"${source_hash}\",\"flags\":${flags_json}}]}")
    string(SHA256 identity "${material}")
    string(APPEND identity_expression "$<$<CONFIG:${configuration}>:${identity}>")
    if(profile EQUAL 1)
      set(runtime_id "${identity}")
    else()
      set(runtime_id "")
    endif()
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/${PP_PACKET_PROFILE_STEM}-${configuration}.json"
      CONTENT "{\n\"schema\":1,\n\"sourceSet\":\"world-static-mesh-flight-controlled-scheduler-v1\",\n\"profile\":${profile},\n\"buildId\":\"${identity}\",\n\"runtimeBuildId\":\"${runtime_id}\",\n\"configuration\":\"${configuration}\",\n\"identityMaterial\":${material},\n\"artifactAuditComplete\":false,\n\"scope\":\"Configured two-source/compiler/strict-flags identity with separate controlled, clock and seven-source references. Not runtime object inspection or complete clock/commit proof.\"\n}\n"
      CONDITION "$<CONFIG:${configuration}>")
  endforeach()
  target_compile_definitions("${target}" PRIVATE
    "PIKAPIKA_CONTROLLED_SCHEDULER_PROFILE=$<IF:$<CONFIG:Release>,${eligible},0>"
    "PIKAPIKA_CONTROLLED_SCHEDULER_BUILD_ID=\"${identity_expression}\""
    "PIKAPIKA_CONTROLLED_SCHEDULER_CONTROLLED_ID=\"${own_controlled_expression}\""
    "PIKAPIKA_CONTROLLED_SCHEDULER_CLOCK_ID=\"${own_clock_expression}\"")
endfunction()
