from pathlib import Path
from hashlib import sha256
import json, subprocess

ROOT = Path.cwd()
OUT = ROOT / 'reports/world-static-mesh-flight-guard-publication/driver-index001'
OUT.mkdir(parents=True, exist_ok=False)
EXACT = {'docs/PLAN.md',
    'reports/2026-11-05-world-static-mesh-flight-guard-xmm-audit.md',
    'reports/world-static-mesh-flight-guard-recipes/check-xmm-operands.py',
    'reports/world-static-mesh-flight-guard-recipes/check-xmm-ci-integration.py',
    'reports/world-static-mesh-flight-guard-publication/xmm-fix.json'}
PREFIXES = [
    'reports/world-static-mesh-flight-guard-public-checks/run-20261105-xmm-operands-002/',
    'reports/world-static-mesh-flight-guard-public-checks/run-20261105-xmm-ci-integration-001/',
    'reports/world-static-mesh-flight-guard-public-checks/run-20261105-xmm-ci-failure-001/',
    'reports/world-static-mesh-flight-guard-public-checks/run-20261105-xmm-fix-status-001/',
    'reports/world-static-mesh-flight-guard-publication/xmm-index001/']

def git(*args, raw=None):
    return subprocess.check_output(['git', *args], cwd=ROOT, input=raw)

paths = [p.decode('utf8') for p in git('diff', '--cached', '--name-only', '-z').split(b'\0') if p]
if not paths or any(p not in EXACT and not any(p.startswith(q) for q in PREFIXES) for p in paths):
    raise ValueError('Unexpected staged path or empty index')
entries = {}
for item in git('ls-files', '--stage', '-z').split(b'\0'):
    if not item:
        continue
    meta, path = item.split(b'\t', 1)
    mode, oid, stage = meta.decode().split()
    if path.decode() in paths:
        if stage != '0':
            raise ValueError('Unmerged index entry')
        entries[path.decode()] = oid
batch = git('cat-file', '--batch', raw=('\n'.join(entries[p] for p in paths) + '\n').encode())
offset, rows = 0, []
for path in paths:
    end = batch.index(b'\n', offset)
    oid, kind, size = batch[offset:end].split()
    size = int(size)
    offset = end + 1
    raw = batch[offset:offset + size]
    offset += size + 1
    if kind != b'blob' or batch[offset - 1:offset] != b'\n':
        raise ValueError('Unexpected batch object')
    working = (ROOT / path).read_bytes()
    if path.startswith('reports/'):
        if raw != working:
            raise ValueError('Proof raw bytes altered by index filter: ' + path)
    elif oid.decode() != git('hash-object', '--path=' + path, '--stdin', raw=working).decode().strip():
        raise ValueError('Working/index mismatch: ' + path)
    rows.append({'path': path, 'oid': oid.decode(), 'workingBytes': len(working),
        'workingSha256': sha256(working).hexdigest(), 'indexBytes': size})
if offset != len(batch):
    raise ValueError('Trailing batch bytes')
result = {'schema': 1, 'status': 'PASS', 'parent': git('rev-parse', 'HEAD').decode().strip(),
    'tree': git('write-tree').decode().strip(), 'files': len(rows),
    'workingRawBytes': sum(x['workingBytes'] for x in rows), 'paths': rows,
    'scope': 'Explicit XMM CI driver correction and actual isolated integration evidence only; separate advance experiment excluded. All proof index bytes equal working raw bytes.'}
(OUT / 'audit-script.py.txt').write_bytes(Path(__file__).read_bytes())
(OUT / 'audit.json').write_text(json.dumps(result, indent=2) + '\n', encoding='utf8')
print(json.dumps({k: result[k] for k in ['status', 'parent', 'tree', 'files', 'workingRawBytes', 'scope']}))
